% Progressive explanation: in Beamer this is \uncover<n->, outside Beamer it
% is a plain include, so the same generated figure works in papers.
\makeatletter
\@ifundefined{insertframenumber}{%
  \newcommand{\wrsReveal}[2]{#2}%
}{%
  \newcommand{\wrsReveal}[2]{\uncover<#1->{#2}}%
}
\makeatother
