% ---------------------------------------------------------------------------
% academic-beamer - semantic macros
%
% The generator calls only these macros with scientific content. All
% presentation decisions (colour, spacing, case, emphasis) live here so the
% template can be restyled without touching generated sources.
% ---------------------------------------------------------------------------

\makeatletter
% semantic status -> palette
\@namedef{wrs@status@new}{wrsPurple}
\@namedef{wrs@status@emerging}{wrsAmber}
\@namedef{wrs@status@plausible}{wrsBlue}
\@namedef{wrs@status@unchanged}{wrsMuted}
\@namedef{wrs@status@strengthened}{wrsGreen}
\@namedef{wrs@status@weakened}{wrsRed}
\@namedef{wrs@status@refuted}{wrsRed}
\@namedef{wrs@status@absent}{wrsFaint}
\newcommand{\wrsStatusColor}[1]{%
  \ifcsname wrs@status@#1\endcsname\csname wrs@status@#1\endcsname\else wrsMuted\fi}

% semantic band role -> palette
\@namedef{wrs@band@measurement}{wrsInkSoft}
\@namedef{wrs@band@observation}{wrsMuted}
\@namedef{wrs@band@interpretation}{wrsBlue}
\@namedef{wrs@band@can}{wrsGreen}
\@namedef{wrs@band@cannot}{wrsRed}
\@namedef{wrs@band@alternative}{wrsMuted}
\@namedef{wrs@band@gap}{wrsAmber}
\newcommand{\wrsBandColor}[1]{%
  \ifcsname wrs@band@#1\endcsname\csname wrs@band@#1\endcsname\else wrsMuted\fi}
\makeatother

% --- text roles -------------------------------------------------------------
\newcommand{\wrsKicker}[1]{{\scriptsize\bfseries\color{wrsBlue}\MakeUppercase{#1}}}
\newcommand{\wrsSection}[1]{{\scriptsize\bfseries\color{wrsMuted}\MakeUppercase{#1}}\par\vspace{0.15em}}
\newcommand{\wrsLead}[1]{{\large\color{wrsInk}#1}}
\newcommand{\wrsBody}[1]{{\normalsize\color{wrsInk}#1}}
\newcommand{\wrsSmall}[1]{{\small\color{wrsInkSoft}#1}}
\newcommand{\wrsQuiet}[1]{{\small\color{wrsMuted}#1}}
\newcommand{\wrsTiny}[1]{{\scriptsize\color{wrsMuted}#1}}
\newcommand{\wrsCitation}[1]{{\scriptsize\color{wrsFaint}#1}}
\newcommand{\wrsMono}[1]{{\ttfamily\small\color{wrsInk}#1}}

% --- semantic elements ------------------------------------------------------
\newcommand{\wrsStatus}[1]{{\small\bfseries\color{\wrsStatusColor{#1}}#1}}
\newcommand{\wrsStatusArrow}{{\small\color{wrsMuted}\,$\rightarrow$\,}}
\newcommand{\wrsTests}[1]{{\scriptsize\color{wrsMuted}tests #1}}

% Leading identifiers are macros, not bare groups: Beamer treats a braced group
% at the start of a frame body as the frame subtitle and would drop it.
\newcommand{\wrsClaimId}[1]{{\large\bfseries\color{wrsInk}#1}}
\newcommand{\wrsDiagId}[1]{{\small\bfseries\color{wrsBlue}#1}}
\newcommand{\wrsClaimTransition}[3]{{\bfseries #1}\quad\wrsStatus{#2}\wrsStatusArrow\wrsStatus{#3}}
\newcommand{\wrsMethodTransition}[3]{%
  {\large\bfseries\color{wrsInkSoft}#1}\ \wrsStatusArrow\ {\large\bfseries\color{wrsBlue}#2}%
  \quad\wrsBody{#3}}

\newenvironment{wrsband}[2]{%
  \par\vspace{0.25em}%
  \begin{beamercolorbox}[wd=\linewidth,sep=0.55em,leftskip=0.45em,rightskip=0.45em]{wrsband}%
  {\scriptsize\bfseries\color{\wrsBandColor{#1}}\MakeUppercase{#2}}\par\vspace{0.18em}%
  \small\color{wrsInkSoft}%
}{%
  \end{beamercolorbox}%
  \vspace{0.25em}%
}

\newcommand{\wrsFigure}[2][\linewidth]{%
  \begin{center}\includegraphics[width=#1,height=0.68\textheight,keepaspectratio]{#2}\end{center}}

% Native TikZ figure generated by the diagram backend (never rasterized).
% #1 optional width; without it the figure keeps its designed size so text is
% not scaled away from the theme hierarchy.
\newcommand{\wrsDiagram}[2][]{%
  \begin{center}%
  \if\relax\detokenize{#1}\relax\input{#2}\else\resizebox{#1}{!}{\input{#2}}\fi
  \end{center}}

\newcommand{\wrsLegendVector}[1]{{\color{wrsRed}$\rightarrow$}\hspace{0.3em}\wrsTiny{#1}}
\newcommand{\wrsLegendReference}[1]{{\color{wrsFaint}\rule[0.3ex]{1.4em}{0.6pt}}\hspace{0.3em}\wrsTiny{#1}}

% --- title page (contents only; the renderer wraps it in a frame) -----------
\newcommand{\wrsTitlePage}[4]{%
  \vfill
  {\color{wrsBlue}\rule{2.4em}{1.6pt}}\par\vspace{0.9em}%
  {\Large\bfseries\color{wrsInk}#2\par}%
  \vspace{0.7em}%
  {\normalsize\color{wrsInkSoft}#3\par}%
  \vspace{1.6em}%
  {\small\color{wrsMuted}#1\par}%
  \vspace{0.25em}%
  {\scriptsize\color{wrsFaint}#4\par}%
  \vfill
}
